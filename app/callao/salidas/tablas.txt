-- 
-- SP: REGISTRAR UNA ENTRADA CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_registrar_entrada_callao (
    IN p_id_producto        INT UNSIGNED,
    IN p_id_tipo_operacion  INT,
    IN p_id_tienda_salida   INT,
    IN p_id_tienda_ingreso  INT,
    IN p_operador           VARCHAR(50),
    IN p_cantidad           INT,
    IN p_id_unidad_medida   INT,
    IN p_entregado_por      VARCHAR(100),
    IN p_registrado_por     VARCHAR(50),
    IN p_observaciones      TEXT
)
BEGIN
    DECLARE v_cant_anterior INT;

    -- Obtener stock actual en la tienda de ingreso (antes del movimiento)
    SELECT IFNULL(cantidad, 0) INTO v_cant_anterior
    FROM existencias_almacen_callao
    WHERE id_producto = p_id_producto AND id_tienda = p_id_tienda_ingreso;

    -- Insertar movimiento con cantidad_anterior calculada
    INSERT INTO movimientos_entrada_callao (
        id_producto, id_tipo_operacion, id_tienda_salida, id_tienda_ingreso,
        operador, cantidad, cantidad_anterior, id_unidad_medida, entregado_por,
        registrado_por, observaciones
    ) VALUES (
        p_id_producto, p_id_tipo_operacion, p_id_tienda_salida, p_id_tienda_ingreso,
        p_operador, p_cantidad, v_cant_anterior, p_id_unidad_medida, p_entregado_por,
        p_registrado_por, p_observaciones
    );

    SET @v_id = LAST_INSERT_ID();

    -- Actualizar existencias sumando la cantidad ingresada
    INSERT INTO existencias_almacen_callao (id_producto, id_tienda, cantidad)
    VALUES (p_id_producto, p_id_tienda_ingreso, p_cantidad)
    ON DUPLICATE KEY UPDATE cantidad = cantidad + p_cantidad;

    SELECT @v_id AS id_movimiento_generado;
END$$

-- 
-- SP: REGISTRAR UNA SALIDA CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_registrar_salida_callao (
    IN p_id_producto        INT UNSIGNED,
    IN p_id_tipo_operacion  INT,
    IN p_nro_comprobante    VARCHAR(50),
    IN p_asesor             VARCHAR(150),
    IN p_cantidad           INT,
    IN p_id_unidad_medida   INT,
    IN p_id_tienda          INT,
    IN p_entregado_por      VARCHAR(100),
    IN p_registrado_por     VARCHAR(50),
    IN p_observaciones      TEXT
)
BEGIN
    DECLARE v_cant_anterior INT;

    -- Obtener stock actual en la tienda (antes de la salida)
    SELECT IFNULL(cantidad, 0) INTO v_cant_anterior
    FROM existencias_almacen_callao
    WHERE id_producto = p_id_producto AND id_tienda = p_id_tienda;

    -- Insertar movimiento con cantidad_anterior calculada
    INSERT INTO movimientos_salida_callao (
        id_producto, id_tipo_operacion, nro_comprobante, asesor,
        cantidad, cantidad_anterior, id_unidad_medida, id_tienda,
        entregado_por, registrado_por, observaciones
    ) VALUES (
        p_id_producto, p_id_tipo_operacion, p_nro_comprobante, p_asesor,
        p_cantidad, v_cant_anterior, p_id_unidad_medida, p_id_tienda,
        p_entregado_por, p_registrado_por, p_observaciones
    );

    SET @v_id = LAST_INSERT_ID();

    -- Actualizar existencias restando la cantidad vendida
    INSERT INTO existencias_almacen_callao (id_producto, id_tienda, cantidad)
    VALUES (p_id_producto, p_id_tienda, -p_cantidad)
    ON DUPLICATE KEY UPDATE cantidad = cantidad - p_cantidad;

    SELECT @v_id AS id_movimiento_generado;
END$$

-- SP: REGISTRAR UNA TRASLADO CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_registrar_traslado_callao (
    IN p_id_producto        INT UNSIGNED,
    IN p_id_tipo_operacion  INT,
    IN p_id_tienda_salida   INT,
    IN p_id_tienda_ingreso  INT,
    IN p_operador           VARCHAR(50),
    IN p_cantidad           INT,
    IN p_id_unidad_medida   INT,
    IN p_entregado_por      VARCHAR(100),
    IN p_registrado_por     VARCHAR(50),
    IN p_observaciones      TEXT
)
BEGIN
    DECLARE v_cant_anterior INT;

    -- Obtener stock actual en la tienda de ingreso (antes del movimiento)
    SELECT IFNULL(cantidad, 0) INTO v_cant_anterior
    FROM existencias_almacen_callao
    WHERE id_producto = p_id_producto AND id_tienda = p_id_tienda_ingreso;

    -- Insertar movimiento con cantidad_anterior calculada
    INSERT INTO movimientos_traslado_callao (
        id_producto, id_tipo_operacion, id_tienda_salida, id_tienda_ingreso,
        operador, cantidad, cantidad_anterior, id_unidad_medida, entregado_por,
        registrado_por, observaciones
    ) VALUES (
        p_id_producto, p_id_tipo_operacion, p_id_tienda_salida, p_id_tienda_ingreso,
        p_operador, p_cantidad, v_cant_anterior, p_id_unidad_medida, p_entregado_por,
        p_registrado_por, p_observaciones
    );

    SET @v_id = LAST_INSERT_ID();

    -- Actualizar existencias sumando la cantidad ingresada
    INSERT INTO existencias_almacen_callao (id_producto, id_tienda, cantidad)
    VALUES (p_id_producto, p_id_tienda_ingreso, p_cantidad)
    ON DUPLICATE KEY UPDATE cantidad = cantidad + p_cantidad;

    SELECT @v_id AS id_movimiento_generado;
END$$


--
-- SP: EDITAR UNA ENTRADA CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_editar_entrada_callao (
    IN p_id_movimiento      INT UNSIGNED,
    IN p_id_producto        INT UNSIGNED,
    IN p_id_tipo_operacion  INT,
    IN p_id_tienda_salida   INT,
    IN p_id_tienda_ingreso  INT,
    IN p_operador           VARCHAR(50),
    IN p_cantidad           INT,
    IN p_id_unidad_medida   INT,
    IN p_entregado_por      VARCHAR(100),
    IN p_registrado_por     VARCHAR(50),
    IN p_observaciones      TEXT,
    IN p_motivo_cambio      TEXT
)
BEGIN
    DECLARE v_cant_ant     INT;
    DECLARE v_tienda_ant   INT;
    DECLARE v_prod_ant     INT UNSIGNED;
    DECLARE v_fecha_orig   DATETIME;
    DECLARE v_nueva_cant_anterior INT;

    -- Obtener datos originales del movimiento
    SELECT cantidad, id_tienda_ingreso, id_producto, fecha_registro
    INTO   v_cant_ant, v_tienda_ant, v_prod_ant, v_fecha_orig
    FROM   movimientos_entrada_callao
    WHERE  id = p_id_movimiento;

    -- Guardar snapshot en historial
    INSERT INTO cambios_entrada_callao (
        id_movimiento_entrada, fecha_movimiento_orig,
        id_producto, id_tipo_operacion, id_tienda_salida, id_tienda_ingreso,
        operador, cantidad, cantidad_anterior, id_unidad_medida,
        entregado_por, registrado_por, observaciones, motivo_cambio
    )
    SELECT id, v_fecha_orig,
           id_producto, id_tipo_operacion, id_tienda_salida, id_tienda_ingreso,
           operador, cantidad, cantidad_anterior, id_unidad_medida,
           entregado_por, registrado_por, observaciones, p_motivo_cambio
    FROM movimientos_entrada_callao
    WHERE id = p_id_movimiento;

    -- Revertir existencia anterior (restar lo que se había sumado)
    UPDATE existencias_almacen_callao
    SET cantidad = cantidad - v_cant_ant
    WHERE id_producto = v_prod_ant AND id_tienda = v_tienda_ant;

    -- Obtener el stock actual en la nueva tienda de ingreso (para cantidad_anterior)
    SELECT IFNULL(cantidad, 0) INTO v_nueva_cant_anterior
    FROM existencias_almacen_callao
    WHERE id_producto = p_id_producto AND id_tienda = p_id_tienda_ingreso;

    -- Actualizar movimiento con los nuevos datos
    UPDATE movimientos_entrada_callao
    SET id_producto         = p_id_producto,
        id_tipo_operacion   = p_id_tipo_operacion,
        id_tienda_salida    = p_id_tienda_salida,
        id_tienda_ingreso   = p_id_tienda_ingreso,
        operador            = p_operador,
        cantidad            = p_cantidad,
        cantidad_anterior   = v_nueva_cant_anterior,
        id_unidad_medida    = p_id_unidad_medida,
        entregado_por       = p_entregado_por,
        registrado_por      = p_registrado_por,
        observaciones       = p_observaciones,
        motivo_cambio       = p_motivo_cambio,
        fecha_actualizacion = NOW()
    WHERE id = p_id_movimiento;

    -- Aplicar nueva existencia (sumar la nueva cantidad)
    INSERT INTO existencias_almacen_callao (id_producto, id_tienda, cantidad)
    VALUES (p_id_producto, p_id_tienda_ingreso, p_cantidad)
    ON DUPLICATE KEY UPDATE cantidad = cantidad + p_cantidad;
END$$


--
-- SP: EDITAR UNA SALIDA CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_editar_salida_callao (
    IN p_id_movimiento      INT UNSIGNED,
    IN p_id_producto        INT UNSIGNED,
    IN p_id_tipo_operacion  INT,
    IN p_nro_comprobante    VARCHAR(50),
    IN p_asesor             VARCHAR(150),
    IN p_cantidad           INT,
    IN p_id_unidad_medida   INT,
    IN p_id_tienda          INT,
    IN p_entregado_por      VARCHAR(100),
    IN p_registrado_por     VARCHAR(50),
    IN p_observaciones      TEXT,
    IN p_motivo_cambio      TEXT
)
BEGIN
    DECLARE v_cant_ant   INT;
    DECLARE v_tienda_ant INT;
    DECLARE v_prod_ant   INT UNSIGNED;
    DECLARE v_fecha_orig DATETIME;
    DECLARE v_nueva_cant_anterior INT;

    -- Obtener datos originales del movimiento
    SELECT cantidad, id_tienda, id_producto, fecha_registro
    INTO   v_cant_ant, v_tienda_ant, v_prod_ant, v_fecha_orig
    FROM   movimientos_salida_callao
    WHERE  id = p_id_movimiento;

    -- Guardar snapshot en historial
    INSERT INTO cambios_salida_callao (
        id_movimiento_salida, fecha_movimiento_orig,
        id_producto, id_tipo_operacion, nro_comprobante, asesor,
        cantidad, cantidad_anterior, id_unidad_medida, id_tienda,
        entregado_por, registrado_por, observaciones, motivo_cambio
    )
    SELECT id, v_fecha_orig,
           id_producto, id_tipo_operacion, nro_comprobante, asesor,
           cantidad, cantidad_anterior, id_unidad_medida, id_tienda,
           entregado_por, registrado_por, observaciones, p_motivo_cambio
    FROM movimientos_salida_callao
    WHERE id = p_id_movimiento;

    -- Revertir existencia anterior (devolver el stock que se había restado)
    UPDATE existencias_almacen_callao
    SET cantidad = cantidad + v_cant_ant
    WHERE id_producto = v_prod_ant AND id_tienda = v_tienda_ant;

    -- Obtener el stock actual en la tienda (para cantidad_anterior)
    SELECT IFNULL(cantidad, 0) INTO v_nueva_cant_anterior
    FROM existencias_almacen_callao
    WHERE id_producto = p_id_producto AND id_tienda = p_id_tienda;

    -- Actualizar movimiento con los nuevos datos
    UPDATE movimientos_salida_callao
    SET id_producto         = p_id_producto,
        id_tipo_operacion   = p_id_tipo_operacion,
        nro_comprobante     = p_nro_comprobante,
        asesor              = p_asesor,
        cantidad            = p_cantidad,
        cantidad_anterior   = v_nueva_cant_anterior,
        id_unidad_medida    = p_id_unidad_medida,
        id_tienda           = p_id_tienda,
        entregado_por       = p_entregado_por,
        registrado_por      = p_registrado_por,
        observaciones       = p_observaciones,
        motivo_cambio       = p_motivo_cambio,
        fecha_actualizacion = NOW()
    WHERE id = p_id_movimiento;

    -- Aplicar nueva salida (restar la nueva cantidad)
    INSERT INTO existencias_almacen_callao (id_producto, id_tienda, cantidad)
    VALUES (p_id_producto, p_id_tienda, -p_cantidad)
    ON DUPLICATE KEY UPDATE cantidad = cantidad - p_cantidad;
END$$

-- SP: CREAR CABECERA DE ABASTECIMIENTO CALLAO
--

CREATE PROCEDURE sp_guardar_abastecimiento_callao (
    IN  p_nombre         VARCHAR(150),
    IN  p_registrado_por VARCHAR(100),
    OUT p_id_cabecera    INT UNSIGNED
)
BEGIN
    INSERT INTO abastecimiento_cabecera_callao (nombre, registrado_por)

    SET p_id_cabecera = LAST_INSERT_ID();

--
-- SP: GUARDAR DETALLES ABASTECIMIENTOS CALLAO
--
DELIMITER $$
CREATE PROCEDURE sp_guardar_abastecimiento_detalle_callao (
    IN p_id_abastecimiento INT UNSIGNED,
    IN p_id_producto       INT UNSIGNED,
    IN p_cant_almacen_oficina         INT,
    IN p_cant_almacen_oficina_docenas         INT,
    IN p_cant_almacen_callao_1_a         INT,
    IN p_cant_almacen_callao_1_b         INT,
    IN p_cant_almacen_callao_2         INT,
    IN p_abastecer_cajas   INT,
    IN p_enviar            ENUM('SI','NO')
)
BEGIN
    DECLARE v_um INT;
    DECLARE v_reg_cant INT;

    -- Obtener unidad de medida y cantidad de reg. cálculo del producto
    SELECT id_unidad_medida_reg, cantidad_reg_calculo
    INTO v_um, v_reg_cant
    FROM productos_abastecimiento_callao
    WHERE id = p_id_producto;

    INSERT INTO abastecimiento_detalle_callao (
        id_abastecimiento, id_producto, cantidad_reg_calculo, id_unidad_medida,
        cant_almacen_oficina, cant_almacen_oficina_docenas, cant_almacen_callao_1_a, cant_almacen_callao_1_b , cant_almacen_callao_2,
        abastecer_cajas, enviar
    ) VALUES (
        p_id_abastecimiento, p_id_producto, v_reg_cant, v_um,
        p_cant_almacen_oficina, p_cant_almacen_oficina_docenas, p_cant_almacen_callao_1_a, p_cant_almacen_callao_1_b, p_cant_almacen_callao_2,
        p_abastecer_cajas, p_enviar
    );
END$$
